b = float(user[2]) # 输入字符串
if op == '+':
  c = a+b
  print(f'{c:.0f}') # 踏马的直接屎山代码堆积
elif op == '-':
  c = a-b
  print(f'{c:.0f}') # 既然不能一步用浮点表示，就直接屎山代码发力
elif op == '*':
  c = a*b
  print(f'{c:.0f}')  #没招
elif op == '/':
  c = a/b
  print(f'{c:.2f}')
else:
  print('无效运算符')

第四题

这题只需要2个for遍历和几个if就可以做出来

#需要你提前知道什么是素数
s = 0#初始化
for i in range(2,1001):#直接从2开始
  is_sushu = True #计算机不知道什么是素数，所以需要写代码来判断
  #因此默认2,1000都是素数
  for j in range (2,i):#从2开始到i
    if i % j == 0: #判断是否为素数
      is_sushu = False #如果不是素数，那么就为false
      break #跳出循环
  if is_sushu == True: 
    print(f'{i:<5}',end="")
    s += i #每行素数个数相加
    if s == 10: #输出了10个素数就换行
      print()
      s = 0 #初始化s=0
